Example: Bronze → Silver Pipeline
Python
# Read Bronze layer
df_bronze = spark.read.format("delta")
    .load("/mnt/bronze/sales/raw")

# Silver transformation
df_silver = df_bronze
    .dropDuplicates(["transaction_id"])
    .filter(col("status") == "active")
    .withColumn("proc_date", current_timestamp())

# Write Silver
df_silver.write.format("delta")
    .mode("overwrite")
    .save("/mnt/silver/sales/clean")
