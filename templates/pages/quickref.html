Delta Lake — Core Commands
SQL
-- Read Delta table
SELECT * FROM catalog.schema.table;

-- Time travel
SELECT * FROM table VERSION AS OF 5;

-- Upsert (Merge)
MERGE INTO target t
USING source s ON t.id = s.id
WHEN MATCHED THEN UPDATE SET *
WHEN NOT MATCHED THEN INSERT *;

-- Optimize & Z-Order
OPTIMIZE table ZORDER BY (customer_id);

-- Vacuum old files (7 day default)
VACUUM table RETAIN 168 HOURS;

-- View history
DESCRIBE HISTORY table;
PySpark — Common Patterns
Python
# Read Delta
df = spark.read.table("catalog.schema.table")

# Write Delta (overwrite)
df.write.format("delta").mode("overwrite")
  .option("overwriteSchema", "true")
  .saveAsTable("catalog.schema.table")

# Upsert via Python API
from delta.tables import DeltaTable
dt = DeltaTable.forName(spark, "catalog.schema.target")
dt.alias("t").merge(source.alias("s"), "t.id = s.id")
  .whenMatchedUpdateAll()
  .whenNotMatchedInsertAll()
  .execute()

DLT — Pipeline Skeleton
Python
import dlt
from pyspark.sql.functions import *

# Bronze — raw ingestion
@dlt.table(name="bronze_sales")
def bronze_sales():
  return spark.readStream.format("cloudFiles")
    .option("cloudFiles.format", "json")
    .load("/mnt/bronze/sales/")

# Silver — with expectations
@dlt.table(name="silver_sales")
@dlt.expect_or_drop("valid_id", "id IS NOT NULL")
def silver_sales():
  return dlt.read_stream("bronze_sales")
    .dropDuplicates(["id"])
    .withColumn("proc_ts", current_timestamp())
